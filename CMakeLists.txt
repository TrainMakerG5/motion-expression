cmake_minimum_required(VERSION 3.24)
project(Expression VERSION 0.1.1 LANGUAGES C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Expression requires 64-bit Windows.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(AVIUTL2_SDK_DIR "${CMAKE_SOURCE_DIR}/third_party/aviutl2_sdk" CACHE PATH "AviUtl2 SDK")
set(QUICKJS_DIR "${CMAKE_SOURCE_DIR}/third_party/quickjs-ng-quickjs-6d46d07" CACHE PATH "QuickJS-ng v0.17.0")
if(NOT EXISTS "${AVIUTL2_SDK_DIR}/module2.h" OR NOT EXISTS "${QUICKJS_DIR}/quickjs.h")
    message(FATAL_ERROR "Run scripts/bootstrap.ps1 first.")
endif()

set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(QJS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${QUICKJS_DIR}" quickjs EXCLUDE_FROM_ALL)
if(MSVC)
    target_compile_options(qjs PRIVATE /W3)
endif()

add_library(expression_engine STATIC src/engine.cpp)
target_include_directories(expression_engine PUBLIC src)
target_link_libraries(expression_engine PUBLIC qjs)

add_library(expression_bindings STATIC src/bindings.cpp)
target_include_directories(expression_bindings PUBLIC src)
if(MSVC)
    target_compile_options(expression_bindings PRIVATE /utf-8 /W4)
endif()

add_library(Expression MODULE src/module.cpp)
target_include_directories(Expression PRIVATE "${AVIUTL2_SDK_DIR}")
target_link_libraries(Expression PRIVATE expression_engine expression_bindings)
target_compile_definitions(Expression PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(Expression PROPERTIES PREFIX "" SUFFIX ".mod2")
if(MSVC)
    target_compile_options(expression_engine PRIVATE /utf-8 /W4)
    target_compile_options(Expression PRIVATE /utf-8 /W4)
endif()

add_library(ExpressionEditor MODULE src/editor.cpp)
target_include_directories(ExpressionEditor PRIVATE "${AVIUTL2_SDK_DIR}")
target_link_libraries(ExpressionEditor PRIVATE expression_bindings user32 gdi32)
target_compile_definitions(ExpressionEditor PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(ExpressionEditor PROPERTIES PREFIX "" SUFFIX ".aux2")
if(MSVC)
    target_compile_options(ExpressionEditor PRIVATE /utf-8 /W4)
endif()

include(CTest)
if(BUILD_TESTING)
    add_library(ExpressionProbe MODULE EXCLUDE_FROM_ALL tests/host_probe.cpp)
    target_include_directories(ExpressionProbe PRIVATE "${AVIUTL2_SDK_DIR}")
    target_compile_definitions(ExpressionProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(ExpressionProbe PRIVATE user32)
    set_target_properties(ExpressionProbe PROPERTIES PREFIX "" SUFFIX ".aux2")
    if(MSVC)
        target_compile_options(ExpressionProbe PRIVATE /utf-8)
    endif()
    add_executable(expression_tests tests/engine_tests.cpp)
    target_link_libraries(expression_tests PRIVATE expression_engine)
    target_link_libraries(expression_tests PRIVATE expression_bindings)
    if(MSVC)
        target_compile_options(expression_tests PRIVATE /utf-8)
    endif()
    add_test(NAME expression_engine COMMAND expression_tests)
    set_tests_properties(expression_engine PROPERTIES TIMEOUT 15)
    add_executable(expression_bridge_tests tests/bridge_tests.cpp)
    target_include_directories(expression_bridge_tests PRIVATE "${AVIUTL2_SDK_DIR}")
    target_compile_definitions(expression_bridge_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(expression_bridge_tests PRIVATE /utf-8 /W4)
    endif()
    foreach(lua_name IN ITEMS lua luaJIT)
        if(EXISTS "${CMAKE_SOURCE_DIR}/tools/aviutl2/${lua_name}.dll")
            add_test(NAME expression_bridge_${lua_name} COMMAND expression_bridge_tests
                "${CMAKE_SOURCE_DIR}/tools/aviutl2/${lua_name}.dll"
                "$<TARGET_FILE:Expression>"
                "${CMAKE_SOURCE_DIR}/scripts/JS制御.anm2")
            set_tests_properties(expression_bridge_${lua_name} PROPERTIES TIMEOUT 15)
        endif()
    endforeach()
endif()

install(TARGETS Expression LIBRARY DESTINATION Script/Expression RUNTIME DESTINATION Script/Expression)
install(TARGETS ExpressionEditor LIBRARY DESTINATION Plugin/Expression RUNTIME DESTINATION Plugin/Expression)
install(FILES scripts/JS制御.anm2 scripts/@Expression.tra2 scripts/expression_track.lua DESTINATION Script/Expression)
install(FILES "${QUICKJS_DIR}/LICENSE" DESTINATION Script/Expression/licenses RENAME QuickJS-ng.txt)
install(FILES "${AVIUTL2_SDK_DIR}/license.txt" DESTINATION Script/Expression/licenses RENAME AviUtl2-SDK.txt)
install(FILES licenses/QuickJS-ng-NOTICES.txt DESTINATION Script/Expression/licenses)
install(FILES README.md LICENSE DESTINATION .)
install(FILES examples/Expression-demo.aup2 DESTINATION examples)
